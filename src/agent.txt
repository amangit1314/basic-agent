"""
The ReAct Agent — The Nervous System
======================================

LEARNING NOTE: What is a ReAct Agent?
--------------------------------------
ReAct = Reasoning + Acting. It's a loop:

    while not done:
        OBSERVE  → What does the agent see?
        THINK    → What should it do next? (LLM decides)
        ACT      → Execute the chosen tool
        UPDATE   → Remember what happened

This is different from a pipeline (fixed steps A→B→C→D) because:
- The agent DECIDES what to do based on what it sees
- It can handle unexpected situations (pagination, redirects, empty pages)
- It can recover from errors (skip a broken link, try a different approach)

ARCHITECTURE:
- The agent owns a Browser (tools) and a GeminiClient (brain)
- State is tracked in an AgentState Pydantic model
- Each iteration logs what happened for transparency
- Safety limit prevents infinite loops

THE HYBRID APPROACH:
--------------------
Pure LLM-driven agents call the LLM on EVERY step, which is expensive.
We use a hybrid: the first few steps (navigate, find_links) are deterministic
(they always happen), but the LLM decides which links to visit and when to stop.
The LLM also handles data extraction. This balances intelligence with cost.
"""

import asyncio
import json
import os
from datetime import datetime
from typing import List, Optional
from urllib.parse import urlparse

from .browser import NoticeBrowser
from .config import (
    DEFAULT_NOTICE_LIMIT,
    MAX_AGENT_STEPS,
    MIN_CONTENT_LENGTH,
    PAGE_DELAY_SECONDS,
    RELEVANT_KEYWORDS,
    logger,
)
from .gemini import LLMClient
from .models import AgentAction, AgentState, AmountInfo, DateInfo, ExtractionResult, NoticeData
from .utils import ExtractionUtils


class NoticeExtractionAgent:
    """
    Autonomous agent for extracting ARC/NPA notices from websites.

    LEARNING NOTE: This is the main class you interact with. Usage:

        agent = NoticeExtractionAgent("https://example.com/notices")
        result = await agent.run()
        # result is an ExtractionResult with all extracted notices

    The agent:
    1. Visits the target URL
    2. Discovers notice links using keyword matching
    3. Visits each notice link (HTML or PDF)
    4. Extracts structured data using Gemini AI
    5. Saves results as JSON

    All decisions are logged so you can trace exactly what the agent did.
    """

    def __init__(
        self,
        url: str,
        keywords: Optional[List[str]] = None,
        max_steps: int = MAX_AGENT_STEPS,
        limit: Optional[int] = DEFAULT_NOTICE_LIMIT,
        provider: str = "gemini",
    ):
        """
        Initialize the agent.

        Args:
            url: Target URL to extract notices from
            keywords: Keywords to match notice links (None = use defaults)
            max_steps: Safety limit on agent steps
            limit: Max number of notices to extract (None = no limit)
            provider: LLM provider — "gemini", "openai", or "claude"
        """
        self.keywords = keywords if keywords is not None else list(RELEVANT_KEYWORDS)
        self.limit = limit

        # Initialize state — the agent's working memory
        self.state = AgentState(
            target_url=url,
            max_steps=max_steps,
        )

        # Initialize tools and brain
        self.browser = NoticeBrowser()
        self.brain = LLMClient(provider=provider)

        # Temporary storage for the last page's content (used between read→extract steps)
        self._last_read_text: str = ""
        self._last_read_url: str = ""
        self._last_read_type: str = "HTML"
        self._current_html: str = ""  # HTML of the current page (for find_links)
        self._last_action_result: str = ""  # Feedback from the last executed action

    # =========================================================================
    # Main ReAct Loop
    # =========================================================================

    async def run(self) -> ExtractionResult:
        """
        Execute the agent's ReAct loop.

        LEARNING NOTE: This is the heart of the agent. Read this method
        carefully — it implements the Observe→Think→Act→Update cycle.

        Returns:
            ExtractionResult containing all extracted notices
        """
        logger.info(f"Agent starting: {self.state.target_url}")
        logger.info(f"Keywords: {len(self.keywords)} | Limit: {self.limit or 'none'}")

        await self.browser.start()

        try:
            while (
                self.state.status != "done"
                and self.state.status != "error"
                and self.state.step_count < self.state.max_steps
            ):
                # ----- OBSERVE -----
                # Build a text description of what the agent currently "sees"
                observation = self._build_observation()

                # ----- THINK -----
                # Ask the brain (Gemini or fallback rules) what to do next
                action = self.brain.decide_next_action(
                    target_url=self.state.target_url,
                    current_url=self.state.current_url,
                    status=self.state.status,
                    step_count=self.state.step_count,
                    max_steps=self.state.max_steps,
                    num_discovered=len(self.state.discovered_links),
                    num_pending=len(self.state.pending_links),
                    num_extracted=len(self.state.extracted_notices),
                    observation=observation,
                    has_unextracted_content=bool(self._last_read_text),
                )

                # Log the decision (this is key for debugging!)
                step_msg = (
                    f"Step {self.state.step_count + 1}: "
                    f"{action.tool}({_summarize_args(action.args)}) "
                    f"— {action.reasoning}"
                )
                self.state.log.append(step_msg)
                logger.info(step_msg)

                # ----- ACT -----
                # Execute the chosen tool
                result = await self._execute_action(action)

                # Store the result for the next observation (FIX #3)
                self._last_action_result = result

                # ----- UPDATE -----
                # Update state based on what happened
                self._update_state(action, result)

                self.state.step_count += 1

                # Rate limiting between page visits
                if action.tool in ("navigate", "read_page"):
                    await asyncio.sleep(PAGE_DELAY_SECONDS)

            # Check if we hit the step limit
            if self.state.step_count >= self.state.max_steps:
                logger.warning(
                    f"Agent hit step limit ({self.state.max_steps}). "
                    f"Extracted {len(self.state.extracted_notices)} notices so far."
                )

            return self._build_result()

        except Exception as e:
            logger.error(f"Agent failed: {e}", exc_info=True)
            self.state.status = "error"
            self.state.log.append(f"FATAL ERROR: {e}")
            return self._build_result()

        finally:
            await self.browser.stop()

    # =========================================================================
    # Observation Builder
    # =========================================================================

    def _build_observation(self) -> str:
        """
        Build a text description of what the agent currently "sees."

        LEARNING NOTE: The observation is the agent's "eyes." It summarizes
        the current state in natural language so the LLM can reason about it.
        A good observation includes:
        - What just happened (last action result)
        - What's available (pending links, current page content)
        - Key metrics (how many notices found, etc.)
        """
        parts = []

        # FIX #3: Include last action result for context
        if self._last_action_result:
            parts.append(f"Last action result: {self._last_action_result}")

        if self.state.status == "starting":
            parts.append("Agent just started. No pages visited yet.")
            parts.append(f"Target URL: {self.state.target_url}")

        elif self.state.status == "exploring":
            if self._current_html:
                parts.append(f"Currently on: {self.state.current_url}")
                parts.append(f"Page loaded with {len(self._current_html)} chars of HTML.")
                if not self.state.discovered_links:
                    parts.append("No links discovered yet — need to run find_links.")
                else:
                    parts.append(
                        f"Discovered {len(self.state.discovered_links)} links, "
                        f"{len(self.state.pending_links)} pending."
                    )

        elif self.state.status == "extracting":
            if self._last_read_text:
                parts.append(
                    f"Just read {self._last_read_url} ({self._last_read_type}, "
                    f"{len(self._last_read_text)} chars)."
                )
                parts.append(f"Content preview: {self._last_read_text[:200]}...")
                parts.append("ACTION NEEDED: Run extract_data to process this content.")
            else:
                parts.append("No unextracted content. Ready to read next link.")
            
            parts.append(f"{len(self.state.pending_links)} links still pending.")
            
            # FIX #1: Show actual next pending link URLs so LLM knows which links to visit
            if self.state.pending_links:
                next_links = self.state.pending_links[:3]  # Show top 3
                parts.append("Next pending links to process:")
                for i, link in enumerate(next_links, 1):
                    parts.append(f"  {i}. {link['url']}")

        if self.state.extracted_notices:
            names = [
                n.company_name or n.borrower_name or "Unknown"
                for n in self.state.extracted_notices[-3:]  # Last 3
            ]
            parts.append(f"Recently extracted: {', '.join(names)}")

        return "\n".join(parts) if parts else "No observations available."

    # =========================================================================
    # Action Executor (the "Act" step)
    # =========================================================================

    async def _execute_action(self, action: AgentAction) -> str:
        """
        Route the agent's chosen action to the appropriate tool.

        LEARNING NOTE: This is the "dispatcher" — it maps tool names to
        actual function calls. The match/case (Python 3.10+) pattern makes
        this very readable. Each tool returns a string result that gets
        logged and fed back into the next observation.
        """
        match action.tool:
            case "navigate":
                return await self._tool_navigate(action.args.get("url", ""))
            case "find_links":
                return self._tool_find_links()
            case "read_page":
                url = action.args.get("url", "")
                # Handle the fallback sentinel — pick next pending link
                if url == "__next_pending__" and self.state.pending_links:
                    url = self.state.pending_links[0]["url"]
                return await self._tool_read_page(url)
            case "extract_data":
                return self._tool_extract_data()
            case "done":
                return "Agent signaled completion."
            case _:
                return f"Unknown tool: {action.tool}"

    # =========================================================================
    # Tool Implementations
    # =========================================================================

    async def _tool_navigate(self, url: str) -> str:
        """
        Tool: Navigate to a URL and get its HTML.

        LEARNING NOTE: This is the first tool the agent uses. It goes to
        the listing page and stores the HTML for link discovery.
        """
        if not url:
            return "ERROR: No URL provided to navigate."

        logger.info(f"  Navigating to: {url}")
        html = await self.browser.get_page_html(url)

        if not html:
            return f"ERROR: Failed to load {url}"

        self._current_html = html
        self.state.current_url = url
        self.state.visited_urls.add(url)
        return f"OK: Loaded {url} ({len(html)} chars of HTML)"

    def _tool_find_links(self) -> str:
        """
        Tool: Scan the current page for notice links.

        LEARNING NOTE: This tool uses the browser's keyword matching to
        find relevant links. It populates both discovered_links (all found)
        and pending_links (to be visited). The limit is applied here.
        """
        if not self._current_html:
            return "ERROR: No page HTML available. Navigate first."

        links = self.browser.discover_links(
            self._current_html,
            self.state.current_url or self.state.target_url,
            self.keywords,
        )

        self.state.discovered_links = links

        # Filter out already-visited URLs and the target URL (listing page)
        links = [
            l for l in links 
            if l["url"] not in self.state.visited_urls
            and l["url"] != self.state.target_url
        ]

        # Apply limit if set
        pending = links
        if self.limit:
            pending = links[:self.limit]
            logger.info(f"  Applied limit: processing {len(pending)} of {len(links)} links")

        self.state.pending_links = pending
        self._current_html = ""  # Free memory

        if not links:
            return "No notice links found on this page."

        # Show top 5 for the observation
        summary = "\n".join(
            f"  - {l['title'][:60]} ({l['type']}) [{', '.join(l['matched_keywords'][:2])}]"
            for l in pending[:5]
        )
        return (
            f"Found {len(links)} links total, processing {len(pending)}.\n"
            f"Top links:\n{summary}"
        )

    async def _tool_read_page(self, url: str) -> str:
        """
        Tool: Visit a URL and read its text content (HTML or PDF).

        LEARNING NOTE: This tool handles the three document types transparently:
        1. HTML page → extracts visible text
        2. PDF link → downloads and parses PDF
        3. Redirect to PDF → detects via content-type and parses

        The content is stored in _last_read_* for the extract_data tool to use.
        
        FIX #2: If URL was already visited, auto-redirect to next pending link
        instead of wasting a step.
        """
        if not url:
            return "ERROR: No URL provided."

        # FIX #2: Auto-redirect to next pending link if URL was already visited
        while url in self.state.visited_urls:
            # Remove the already-visited URL from pending list if present
            self.state.pending_links = [
                l for l in self.state.pending_links if l["url"] != url
            ]
            if not self.state.pending_links:
                return f"SKIP: Already visited {url} and no more pending links."
            # Pick the next pending link
            url = self.state.pending_links[0]["url"]
            logger.info(f"  Auto-redirecting to next pending: {url}")

        logger.info(f"  Reading: {url}")
        text, doc_type, nested_links = await self.browser.get_page_text(url)

        self.state.visited_urls.add(url)

        # Remove this link from pending
        self.state.pending_links = [
            l for l in self.state.pending_links if l["url"] != url
        ]

        if not text or len(text) < MIN_CONTENT_LENGTH:
            return f"SKIP: Insufficient content from {url} ({len(text)} chars)"

        # Store for extraction
        self._last_read_text = text
        self._last_read_url = url
        self._last_read_type = doc_type
        self.state.last_read_nested_links = nested_links or []

        # Check for nested links (PDFs)
        nested_msg = ""
        if nested_links:
            pdf_links = [l for l in nested_links if l.lower().endswith(".pdf")]
            if pdf_links:
                nested_msg = f"\nFound {len(pdf_links)} nested PDF links. LLM should consider reading these: {', '.join(pdf_links[:3])}..."

        return (
            f"OK: Read {url} ({doc_type}, {len(text)} chars). "
            f"Preview: {text[:150]}...{nested_msg}"
        )

    def _tool_extract_data(self) -> str:
        """
        Tool: Extract structured data from the last read page.

        LEARNING NOTE: This is a two-phase extraction:
        1. Primary: Gemini AI extracts structured JSON
        2. Fallback: Regex patterns fill in missing fields

        The fallback ensures we get SOMETHING even if the LLM misses fields.
        This is a common pattern: LLM for the heavy lifting, regex for reliability.
        """
        if not self._last_read_text:
            return "ERROR: No text to extract from. Read a page first."

        # Phase 1: AI extraction
        notice = self.brain.extract_notice_data(
            text=self._last_read_text,
            source_url=self._last_read_url,
            source_type=self._last_read_type,
        )
        # Populate nested links from state
        notice.nested_links = self.state.last_read_nested_links

        # Phase 2: Regex fallbacks for missing fields
        notice = self._apply_regex_fallbacks(notice, self._last_read_text)

        # Store the result
        self.state.extracted_notices.append(notice)

        # Clear temporary storage
        self._last_read_text = ""
        self._last_read_url = ""
        self._last_read_type = "HTML"

        name = notice.company_name or notice.borrower_name or "Unknown"
        return (
            f"OK: Extracted notice for '{name}' — "
            f"{len(notice.dates)} dates, {len(notice.amounts)} amounts"
        )

    # =========================================================================
    # Regex Fallbacks
    # =========================================================================

    def _apply_regex_fallbacks(self, notice: NoticeData, text: str) -> NoticeData:
        """
        Fill in missing fields using regex patterns.

        LEARNING NOTE: LLMs are great at understanding context but sometimes
        miss specific patterns. Regex is the opposite — it's rigid but reliable
        for known patterns. By combining both, we get the best of both worlds.
        """
        # Fallback: dates
        regex_dates = ExtractionUtils.extract_dates(text)
        if not notice.dates:
            for i, d in enumerate(regex_dates[:5]):
                label = "auction_date" if i == 0 else f"date_{i + 1}"
                notice.dates.append(DateInfo(label=label, value=d))
        
        # Populate explicit auction_date if missing
        if not notice.auction_date and regex_dates:
            notice.auction_date = regex_dates[0]

        # Fallback: reserve price
        has_reserve = any("reserve" in a.label.lower() for a in notice.amounts)
        regex_reserve = ExtractionUtils.extract_reserve_price(text)
        if not has_reserve and regex_reserve:
            notice.amounts.append(AmountInfo(label="reserve_price", value=regex_reserve))
        
        # Populate explicit reserve_price if missing
        if not notice.reserve_price and regex_reserve:
            notice.reserve_price = regex_reserve

        # Fallback: due amount
        has_dues = any(
            "due" in a.label.lower() or "outstanding" in a.label.lower()
            for a in notice.amounts
        )
        regex_dues = ExtractionUtils.extract_due_amount(text)
        if not has_dues and regex_dues:
            notice.amounts.append(AmountInfo(label="total_dues", value=regex_dues))

        # Populate explicit due_amount if missing
        if not notice.due_amount and regex_dues:
            notice.due_amount = regex_dues

        # Fallback: notice type
        if not notice.notice_type:
            notice.notice_type = ExtractionUtils.determine_notice_type(text)

        return notice

    # =========================================================================
    # State Updates
    # =========================================================================

    def _update_state(self, action: AgentAction, result: str) -> None:
        """
        Update the agent's status based on what just happened.

        LEARNING NOTE: State transitions are the "control flow" of the agent.
        Instead of if/else chains, we use status transitions:
        starting → exploring → extracting → done

        The agent can also loop (extracting → extracting) when processing
        multiple notices.
        """
        if action.tool == "done":
            self.state.status = "done"

        elif action.tool == "navigate":
            if result.startswith("OK"):
                self.state.status = "exploring"
            else:
                self.state.status = "error"

        elif action.tool == "find_links":
            if self.state.pending_links:
                self.state.status = "extracting"
            else:
                self.state.status = "done"

        elif action.tool == "read_page":
            # Stay in extracting — we still need to extract data from what we read
            self.state.status = "extracting"

        elif action.tool == "extract_data":
            # Check if there are more links to process
            if not self.state.pending_links:
                self.state.status = "done"
            # else stay in "extracting"

    # =========================================================================
    # Result Builder
    # =========================================================================

    def _build_result(self) -> ExtractionResult:
        """
        Build the final output from the agent's state.

        LEARNING NOTE: The ExtractionResult wraps everything with metadata.
        This is saved to JSON in the results/ folder.
        """
        return ExtractionResult(
            source_url=self.state.target_url,
            agent_steps_taken=self.state.step_count,
            total_notices_found=len(self.state.extracted_notices),
            notices=self.state.extracted_notices,
        )

    # =========================================================================
    # Save to File
    # =========================================================================

    def save_results(self, result: ExtractionResult) -> str:
        """
        Save extraction results to a JSON file in results/ folder.

        File naming: {domain}_{timestamp}.json
        Example: idbicapital_com_20260209_143022.json

        LEARNING NOTE: We use Pydantic's model_dump() to serialize.
        This converts all our typed models into plain dicts that json.dump()
        can handle. No manual serialization needed!
        """
        results_dir = os.path.join(os.path.dirname(os.path.dirname(__file__)), "results")
        os.makedirs(results_dir, exist_ok=True)

        domain = urlparse(result.source_url).netloc.replace(".", "_")
        timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")
        filepath = os.path.join(results_dir, f"{domain}_{timestamp}.json")

        with open(filepath, "w", encoding="utf-8") as f:
            # Exclude raw_text_snippet to save space/cost
            json.dump(result.model_dump(exclude={"notices": {"__all__": {"raw_text_snippet"}}}), f, indent=2, ensure_ascii=False)

        logger.info(f"Results saved to: {filepath}")
        return filepath


# =============================================================================
# Helper
# =============================================================================

def _summarize_args(args: dict) -> str:
    """Shorten args dict for logging (don't print huge text blocks)."""
    parts = []
    for k, v in args.items():
        if isinstance(v, str) and len(v) > 60:
            v = v[:57] + "..."
        parts.append(f"{k}={v}")
    return ", ".join(parts) if parts else ""
